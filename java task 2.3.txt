
import java.util.*;

public class Main {

    public static int removeDuplicates(int[] nums) {
        if (nums.length == 0) {
            return 0;
        }

        int k = 1;

        for (int i = 1; i < nums.length; i++) {
            if (nums[i] != nums[i - 1]) {
                nums[k] = nums[i];
                k++;
            }
        }

        return k;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Read array size
        int n = sc.nextInt();

        // Read sorted array
        int[] nums = new int[n];

        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }

        // Remove duplicates
        int k = removeDuplicates(nums);

        // Print number of unique elements
        System.out.println(k);

        // Print unique elements
        for (int i = 0; i < k; i++) {
            System.out.print(nums[i]);

            if (i < k - 1) {
                System.out.print(" ");
            }
        }

        sc.close();
    }
}
