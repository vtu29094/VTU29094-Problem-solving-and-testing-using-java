```java
import java.util.*;

public class Main {

    public static int removeElement(int[] nums, int val) {
        int k = 0;

        for (int i = 0; i < nums.length; i++) {
            if (nums[i] != val) {
                nums[k] = nums[i];
                k++;
            }
        }

        return k;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Read array size
        int n = sc.nextInt();

        // Read array
        int[] nums = new int[n];

        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }

        // Read value to remove
        int val = sc.nextInt();

        // Remove elements
        int k = removeElement(nums, val);

        // Print k
        System.out.println(k);

        // Print remaining elements
        for (int i = 0; i < k; i++) {
            System.out.print(nums[i]);

            if (i < k - 1) {
                System.out.print(" ");
            }
        }

        sc.close();
    }
}
```
